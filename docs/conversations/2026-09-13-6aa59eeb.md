```bash
import sys
import time
sys.path.insert(0, "/home/aria")
from aria_sensors import SensorMonitor, format_human
from adafruit_bme280 import basic

m = SensorMonitor()
old_mode = None
try:
    m.initialize()
    if m.bme is None:
        raise RuntimeError("BME khởi tạo lỗi; dừng thử")
    old_mode = m.bme.mode
    m.bme.mode = basic.MODE_NORMAL
    time.sleep(1)
    for _ in range(10):
        start = time.perf_counter()
        sample = m.sample()
        ms = (time.perf_counter() - start) * 1000
        print(f"{ms:.1f} ms | {format_human(sample)}", flush=True)
        time.sleep(3)
except KeyboardInterrupt:
    pass
finally:
    try:
        if m.bme is not None and old_mode is not None:
            m.bme.mode = old_mode
    finally:
        m.cleanup()
PY
```
